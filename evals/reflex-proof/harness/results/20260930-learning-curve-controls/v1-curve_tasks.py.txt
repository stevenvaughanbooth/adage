"""Prospective learning-curve tasks; the historical showcase registry stays unchanged."""
from copy import deepcopy
from pathlib import Path
import sys

sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "showcase"))
import fam_python


def simple(vid, cls, desc, valid, invalid, check):
    message = f"Must satisfy {cls}."
    return dict(id=vid, cls=cls, sig=f"{cls}(*, error=None)", args="", desc=desc,
                msg=message, keys=["input"], formatted=message, repr_arg="",
                valid=valid, invalid=invalid, custom=("invalid: {input}", "invalid: " + str(eval(invalid[0], {"__builtins__": {}}, {}))),
                init=[], check=check)


# First 12 are reused task definitions, but require NEW native sessions in this experiment.
# Four final definitions have never been used as training tasks in the earlier experiments.
VARIANTS = deepcopy(fam_python.MM_VARIANTS) + [
    simple("palindrome", "Palindrome", "which accepts strings equal to their reversal (case-sensitive, including empty strings).",
           ["'level'", "''", "'été'"], ["'Level'", "'ab'"], "value == value[::-1]"),
    simple("strictly-increasing", "StrictlyIncreasing", "which accepts numeric sequences whose adjacent elements strictly increase; empty and single-element sequences pass.",
           ["[1, 3, 8]", "[]", "[2]"], ["[1, 1]", "[2, 1]"], "all(a < b for a, b in zip(value, value[1:]))"),
    simple("printable", "Printable", "which accepts non-empty strings containing only printable characters according to Python str.isprintable().",
           ["'hello world'", "'é'"], ["''", "'a\\nb'", "'\\t'"], "value.isprintable()"),
    simple("odd", "Odd", "which accepts integers whose remainder modulo two is one (including negative odd integers).",
           ["3", "-3", "101"], ["0", "4", "-8"], "value % 2 == 1"),
    simple("finite", "Finite", "which accepts finite real numbers; NaN and positive or negative infinity fail.",
           ["0", "-1.25", "1e300"], ["1e309", "-1e309", "float('nan')"], "__import__('math').isfinite(value)"),
    simple("no-adjacent-duplicates", "NoAdjacentDuplicates", "which accepts sequences with no equal adjacent elements; nonadjacent repetition is allowed, as are empty and single-element sequences.",
           ["[1, 2, 1]", "[]", "[[1], [2], [1]]"], ["[1, 1]", "'abb'", "[[1], [1]]"], "all(a != b for a, b in zip(value, value[1:]))"),
    dict(id="max-utf8-bytes", cls="MaxUtf8Bytes", sig="MaxUtf8Bytes(max, *, error=None)", args="4",
         desc="which accepts strings whose UTF-8 encoding is at most `max` bytes long (including empty strings).",
         msg="Must occupy at most {max} UTF-8 bytes.", keys=["input", "max"],
         formatted="Must occupy at most 4 UTF-8 bytes.", repr_arg="max=4",
         valid=["'abcd'", "'éé'", "''", "'😀'"], invalid=["'abcde'", "'ééa'", "'😀a'"],
         custom=("{input} exceeds {max} bytes", "abcde exceeds 4 bytes"), init=["max"],
         check="len(value.encode('utf-8')) <= self.max"),
]


def prompt(variant):
    # Remaining examples are private boundary cases, not copied into the task prompt.
    return fam_python.lab.mm_prompt(dict(variant, valid=variant["valid"][:1], invalid=variant["invalid"][:1]))


FAMILY = dict(fam_python.MM, id="mm-curve", variants=VARIANTS, prompt=prompt)
TASKS = ["mm-curve--" + v["id"] for v in VARIANTS]
TRAIN, HELDOUT = TASKS[:15], TASKS[15:]


def install(registry):
    """Register only in the dedicated experiment process, never alter historical task ordering."""
    registry.FAMILIES[FAMILY["id"]] = FAMILY
